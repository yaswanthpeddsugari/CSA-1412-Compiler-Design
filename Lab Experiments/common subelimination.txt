#include <stdio.h>
#include <string.h>

int main()
{
    char a[20], b[20];

    printf("Enter first expression: ");
    scanf("%s", a);

    printf("Enter second expression: ");
    scanf("%s", b);

    if (strcmp(a, b) == 0)
        printf("Common expression found\n");
    else
        printf("No common expression\n");

    return 0;
}
