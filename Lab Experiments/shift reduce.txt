#include <stdio.h>

int main()
{
    char input[20];
    int i = 0;

    printf("Enter input (i+i): ");
    scanf("%s", input);

    printf("\nStack\tInput\tAction\n");

    while (input[i] != '\0')
    {
        printf("%c\t%s\tShift\n",
               input[i], input+i);
        i++;
    }

    printf("E\t$\tReduce\n");
    printf("E\t$\tAccept\n");

    return 0;
}
