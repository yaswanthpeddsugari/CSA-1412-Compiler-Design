#include <stdio.h>

int main()
{
    char s[500];
    int i = 0, count = 0;

    printf("Enter C code:\n");
    scanf(" %[^\n]", s);

    while (s[i] != '\0')
    {
        if (s[i] == '/' && s[i+1] == '/')
            count++;

        if (s[i] == '/' && s[i+1] == '*')
            count++;

        i++;
    }

    printf("Comment count = %d\n", count);

    return 0;
}
